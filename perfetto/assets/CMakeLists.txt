cmake_minimum_required(VERSION 3.16)
project(perfetto_skill_probe LANGUAGES CXX)
set(PERFETTO_SDK_DIR "" CACHE PATH "Directory containing release perfetto.h and perfetto.cc")
if(NOT IS_ABSOLUTE "${PERFETTO_SDK_DIR}" OR
   NOT EXISTS "${PERFETTO_SDK_DIR}/perfetto.h" OR
   NOT EXISTS "${PERFETTO_SDK_DIR}/perfetto.cc")
  message(FATAL_ERROR "Set PERFETTO_SDK_DIR to the absolute extracted Perfetto C++ SDK directory")
endif()
find_package(Threads REQUIRED)
add_library(perfetto_sdk STATIC "${PERFETTO_SDK_DIR}/perfetto.cc")
target_include_directories(perfetto_sdk PUBLIC "${PERFETTO_SDK_DIR}")
target_compile_features(perfetto_sdk PUBLIC cxx_std_17)
target_link_libraries(perfetto_sdk PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(ANDROID)
  target_link_libraries(perfetto_sdk PUBLIC log)
endif()
if(WIN32)
  target_compile_definitions(perfetto_sdk PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(perfetto_sdk PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(perfetto_sdk PRIVATE /bigobj /permissive-)
endif()
add_executable(perfetto-skill-probe track-event-smoke.cc)
target_link_libraries(perfetto-skill-probe PRIVATE perfetto_sdk)
if(MSVC)
  target_compile_options(perfetto-skill-probe PRIVATE /permissive-)
endif()
