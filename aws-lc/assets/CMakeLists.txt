cmake_minimum_required(VERSION 3.20)
project(awslc_consumer_probe LANGUAGES C)

set(AWSLC_PREFIX "" CACHE PATH "Absolute selected AWS-LC install prefix")
option(AWSLC_DIST_PKG "Use suffixed distribution-package libraries and headers" OFF)
if(NOT IS_ABSOLUTE "${AWSLC_PREFIX}")
  message(FATAL_ERROR "Set -DAWSLC_PREFIX to the absolute AWS-LC install prefix")
endif()

if(AWSLC_DIST_PKG)
  set(awslc_include "${AWSLC_PREFIX}/include/aws-lc")
  set(awslc_crypto_name crypto-awslc)
  set(awslc_ssl_name ssl-awslc)
else()
  set(awslc_include "${AWSLC_PREFIX}/include")
  set(awslc_crypto_name crypto)
  set(awslc_ssl_name ssl)
endif()
if(NOT EXISTS "${awslc_include}/openssl/aead.h")
  message(FATAL_ERROR "Selected prefix does not contain AWS-LC AEAD headers")
endif()
find_library(awslc_crypto NAMES ${awslc_crypto_name}
  PATHS "${AWSLC_PREFIX}/lib" "${AWSLC_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
find_library(awslc_ssl NAMES ${awslc_ssl_name}
  PATHS "${AWSLC_PREFIX}/lib" "${AWSLC_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
message(STATUS "AWS-LC include: ${awslc_include}")
message(STATUS "AWS-LC crypto: ${awslc_crypto}")
message(STATUS "AWS-LC ssl: ${awslc_ssl}")
add_executable(awslc-smoke smoke.c)
target_compile_features(awslc-smoke PRIVATE c_std_99)
target_include_directories(awslc-smoke PRIVATE "${awslc_include}")
target_link_libraries(awslc-smoke PRIVATE "${awslc_ssl}" "${awslc_crypto}" Threads::Threads)
option(AWSLC_WINDOWS_SHARED "Selected Windows libraries are DLL import libraries" OFF)
if(WIN32 AND AWSLC_WINDOWS_SHARED)
  target_compile_definitions(awslc-smoke PRIVATE BORINGSSL_SHARED_LIBRARY)
endif()
